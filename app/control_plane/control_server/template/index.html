01

Platform Overview

Signal Layer

Kernel-first event capture

Host and container signals are captured from process, file, and network activity with low overhead and strict budget awareness.

  • Process spawn and lineage visibility
  • Filesystem access and write intent
  • Outbound flow and destination context
Reasoning Layer

Rule correlation and scoring

OIL rules correlate events in windows, compare them to baselines, and assign risk with transparent score math.

  • Cross-domain joins in one rule flow
  • Time-window chaining (`within`)
  • Facts for follow-on detections
Response Layer

Actionable enforcement outputs

Rules can alert, snapshot key artifacts, isolate risky hosts, and enforce outbound controls when score thresholds are crossed.

  • Tiered `alert` severity
  • `snapshot` for fast investigation
  • `isolate` and `block egress` controls

OIL Rule Examples

Exfiltration

Credential and outbound chaining

Pairs credential access with outbound traffic to catch suspicious process-driven extraction paths.

Containers

Interactive and privileged runtime risk

Flags interactive shells and privileged container starts, with context from workload metadata and identity sessions.

Integrity

Binary and root SSH tampering

Detects suspicious writes to executable directories and sensitive root SSH paths with baseline-aware scoring.

oilc/src/rules/credential_access_followed_by_egress.oil
set credential_targets = [
  "/etc/shadow",
  "/root/.ssh/id_rsa",
  "/home/*/.ssh/id_rsa",
  "~/.aws/credentials",
  "~/.kube/config"
]

rule "credential_access_followed_by_egress" {
  from endpoint.file, endpoint.process, network.flow

  correlate
    file.open as f by process as p
    with network.connect as n on n.process_id == p.id

  where
    f.path matches credential_targets
    and f.mode in ["read", "read_write"]
    and n.direction == "outbound"

  within 10m

  let
    unusual_dest = n.dest.domain not in host(p.host_id).baseline.domains
    suspicious_proc = p.name in ["python", "node", "perl", "ruby", "bash", "curl"]
    high_exfil = n.bytes_out > 100000

  score
    65
    +10 if unusual_dest
    +10 if suspicious_proc
    +15 if high_exfil
    +10 if n.dest.reputation in ["suspicious", "malicious"]

  emit
    fact host.possible_credential_exfil(host.id) expires 8h

  respond
    if score >= 85 {
      alert critical
      isolate host
      snapshot p, f, n
      open case "Credential access followed by egress"
    } else {
      alert high
    }
}
oilc/src/rules/shell_outbound_to_bad_reputation.oil
fact host.shell_bad_egress(host_id) expires 4h

rule "shell_outbound_to_bad_reputation" {
  from endpoint.process, network.flow

  correlate
    process.spawn as p
    with network.connect as n on n.process_id == p.id

  where
    p.name in shells
    and n.direction == "outbound"
    and n.dest.reputation in bad_reputation

  let
    high_exfil = n.bytes_out > 500000
    public_dest = n.dest.is_internal == false

  score
    65
    +10 if public_dest
    +15 if high_exfil

  emit
    fact host.shell_bad_egress(host.id) expires 4h

  respond
    if score >= 80 {
      alert critical
      snapshot p, n
      open case "Shell outbound connection to bad reputation destination"
    } else {
      alert high
    }
}
oilc/src/rules/shell_spawn_in_container.oil
set shell_binaries = ["sh", "bash", "dash", "zsh", "fish"]
set approved_debug_images = []

rule "interactive_shell_in_container" {
  from endpoint.process, container.runtime

  correlate
    process.spawn as p
    with container.context as c on c.container_id == p.container_id

  where
    is_shell(p)
    and c.exists == true
    and c.image.repo not in approved_debug_images

  let
    prod = c.tags contains "prod"
    k8s = c.orchestrator == "kubernetes"
    interactive = p.tty == true
    suspicious_parent = p.parent.name not in ["pause", "tini", "entrypoint.sh"]

  score
    45
    +20 if prod
    +15 if interactive
    +10 if suspicious_parent
    +10 if k8s
    +10 if p.user.name == "root"

  emit
    fact container.interactive_shell(c.container_id) expires 2h

  respond
    if score >= 75 {
      alert critical
      snapshot p, c
      open case "Interactive shell in container"
    } else {
      alert medium
    }
}
oilc/src/rules/launch_of_priviledge_container.oil
rule "privileged_container_started" {
  from container.runtime, k8s.admission, identity.session

  correlate
    container.start as c
    with session.use as s on s.user_id == c.initiator.user_id

  where
    c.security.privileged == true

  let
    host_ns = c.namespaces.host_pid or c.namespaces.host_network or c.namespaces.host_ipc
    dangerous_caps = count(c.capabilities.added) > 0
    prod = c.labels.environment == "production"
    initiator_new_geo = s.country not in user(s.user_id).baseline.countries

  score
    60
    +15 if host_ns
    +15 if dangerous_caps
    +15 if prod
    +10 if c.image.tag == "latest"
    +10 if initiator_new_geo

  emit
    fact container.high_risk(c.container_id) expires 24h

  respond
    if score >= 85 {
      alert critical
      open case "Privileged container launched"
      snapshot c, s
    } else {
      alert high
    }
}
oilc/src/rules/modify_binary_directories.oil
set executable_dirs = [
  "/bin",
  "/sbin",
  "/usr/bin",
  "/usr/sbin",
  "/usr/local/bin",
  "/usr/local/sbin"
]

rule "binary_directory_modification" {
  from endpoint.file, endpoint.process

  correlate
    file.write as f by process as p

  where
    f.path under executable_dirs

  let
    package_mgmt = p.name in ["apt", "apt-get", "dpkg", "yum", "dnf", "rpm"]
    interactive = p.parent.name in ["bash", "sh", "zsh"]
    suspicious_path = p.binary.path under ["/tmp", "/var/tmp", "/dev/shm"]

  score
    60
    -30 if package_mgmt
    +15 if interactive
    +20 if suspicious_path
    +10 if p.user.name != "root"

  require
    score >= 40

  emit
    fact host.binary_tamper(host.id) expires 24h

  respond
    if score >= 80 {
      alert critical
      snapshot p, f
      open case "Executable directory modification"
    } else {
      alert medium
    }
}
oilc/src/rules/root_ssh_write_by_non_root_process.oil
fact host.root_ssh_write(host_id) expires 6h

rule "root_ssh_write_by_non_root_process" {
  from endpoint.file, endpoint.process

  correlate
    file.write as f by process as p

  where
    f.path under "/root/.ssh"
    and p.uid != 0

  let
    suspicious_proc = p.name not in ["sshd", "scp", "rsync"]
    stealthy_target = f.path in ["/root/.ssh/authorized_keys", "/root/.ssh/id_rsa"]

  score
    60
    +15 if suspicious_proc
    +20 if stealthy_target

  emit
    fact host.root_ssh_write(host.id) expires 6h

  respond
    if score >= 80 {
      alert critical
      snapshot p, f
      open case "Write under /root/.ssh by non-root process"
    } else {
      alert high
    }
}
